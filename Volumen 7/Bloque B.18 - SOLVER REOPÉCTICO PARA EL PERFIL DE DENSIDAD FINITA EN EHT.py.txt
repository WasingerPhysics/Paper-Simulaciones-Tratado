import numpy as np
import matplotlib.pyplot as plt

# Parámetros constitutivos del tapón viscoelástico
mu_caputo = 0.035  # Orden fraccionario de retención de la esponja (1 - ns)
densidad_saturacion_planck = 8.5  # Límite elástico de compactación granular de la Persistencia

# Radio relativo respecto al horizonte de sucesos (R=1 es la garganta del poro)
radio_r = np.linspace(0.01, 3.5, 500)

# 1. Perfil de Densidad de la Relatividad General Clásica (alpha = 1)
densidad_clasica_singularidad = 1.0 / (radio_r ** 2)

# 2. Perfil de Densidad de la Persistencia (Tapón Granular con Memoria)
densidad_tapon_persistencia = densidad_saturacion_planck / (1.0 + (radio_r ** (2.0 - mu_caputo)))

# 3. Simulación de datos de perfiles de densidad (Datos Sombras EHT)
np.random.seed(22)
ruido_resolucion_eht = np.random.uniform(0.1, 0.4, 500)
datos_crudos_sombras = densidad_tapon_persistencia + np.random.normal(0, ruido_resolucion_eht * 0.2, 500)
datos_crudos_sombras = np.clip(datos_crudos_sombras, 0.05, 12.0)

# 4. Renderizado del Gráfico de Estructura Profunda con blindaje estricto de comillas
plt.figure(figsize=(9, 6))
plt.scatter(radio_r, datos_crudos_sombras, color='darkred', alpha=0.25, s=8, 
            label="Densidad Involucrada de Masa-Energía (Datos Sombras EHT)")
plt.plot(radio_r, densidad_tapon_persistencia, 'r-', linewidth=2.5, 
         label=f"Horizonte de Retención de Caputo (mu={mu_caputo})") # Corrección gramatical y comillas dobles
plt.plot(radio_r, densidad_clasica_singularidad, 'b--', linewidth=1.5, label="Singularidad Relativista Tradicional (alpha=1)")

plt.title('Disolución de la Singularidad: Perfil Reológico del Tapón Viscoelástico', fontsize=11, fontweight='bold')
plt.xlabel('Distancia Radial al Centro del Objeto (R) [Radios de Horizonte]')
plt.ylabel('Densidad Efectiva de Masa-Energía Relativa')
plt.xlim(0, 3.5)
plt.ylim(0, 11)
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend()
plt.tight_layout()
plt.show()
