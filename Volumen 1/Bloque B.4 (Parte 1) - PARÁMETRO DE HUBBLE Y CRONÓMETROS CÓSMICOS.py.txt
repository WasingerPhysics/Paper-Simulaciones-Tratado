import numpy as np
import matplotlib.pyplot as plt

# Parámetros constitutivos derivados del lecho viscoelástico con memoria
alpha_opt, H0_opt = 0.0096, 65.32
z_linea = np.linspace(0.01, 2.5, 100)
H_teorico = H0_opt * np.sqrt(0.315*(1+z_linea)**3 + 0.685*(1+z_linea)**(3*alpha_opt))

# Submuestra Clave Seleccionada de 9 puntos estratégicos sobre el catálogo de Cronómetros Cósmicos
z_cc = np.array([0.09, 0.17, 0.27, 0.40, 0.48, 0.90, 1.30, 1.75, 1.97])
H_cc = np.array([69.0, 83.0, 77.0, 82.0, 85.0, 117.0, 168.0, 202.0, 224.0])
err_cc = np.array([12.0, 8.0, 14.0, 11.0, 12.0, 23.0, 17.0, 40.0, 38.0])

# Renderizado gráfico con blindaje estricto de strings para evitar errores de sintaxis
plt.figure(figsize=(8, 5.5))
plt.errorbar(z_cc, H_cc, yerr=err_cc, fmt='o', color='navy', capsize=4, label="Datos H(z) Reales (CC)")
plt.plot(z_linea, H_teorico, 'r-', linewidth=2.5, label="Persistencia Hidrosocial (α=0.0096)")

plt.title('B. Parámetro de Hubble H(z) frente a Cronómetros Cósmicos')
plt.xlabel('Corrimiento al rojo (z)')
plt.ylabel('H(z) [km/s/Mpc]')
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend()
plt.tight_layout()
plt.show()
