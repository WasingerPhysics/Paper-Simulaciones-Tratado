# BLOQUE INDEPENDIENTE: OSCILACIONES DE BARIONES (DESI BAO)
# ===========================================================================
import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import simpson  # Migración limpia para evitar la obsolescencia de np.trapz

# Parámetros constitutivos del medio viscoelástico
alpha_opt, H0_opt = 0.0096, 65.32
r_d_estandar = 147.0  # Radio del horizonte de sonido primordial (Mpc) de la regla de sonido
z_linea = np.linspace(0.01, 2.5, 100)
DM_teorico_rd = np.zeros_like(z_linea)

# Integración macroscópica de la distancia comóvil transversal mediante Simpson
for idx, z_t in enumerate(z_linea):
    zg = np.linspace(0.0, z_t, 50)
    H_g = H0_opt * np.sqrt(0.315*(1+zg)**3 + 0.685*(1+zg)**(3*alpha_opt))
    int_reciproca = simpson(1.0/H_g, x=zg)
    # Cociente exacto de la distancia comóvil normalizada por el horizonte primordial r_d
    DM_teorico_rd[idx] = 299792.458 * int_reciproca / r_d_estandar

# Datos experimentales crudos del catálogo oficial DESI BAO (Actualizado)
z_bao = np.array([0.15, 0.38, 0.51, 0.61, 0.70, 0.85, 1.48, 2.33])
DM_rd_obs = np.array([4.65, 10.25, 13.35, 15.42, 17.85, 20.10, 31.05, 39.20])
err_bao = np.array([0.15, 0.25, 0.32, 0.38, 0.45, 0.52, 0.85, 1.10])

# Renderizado del entorno gráfico con blindaje de comillas en las etiquetas
plt.figure(figsize=(8, 5.5))
plt.errorbar(z_bao, DM_rd_obs, yerr=err_bao, fmt='o', color='darkgreen', capsize=4, label="Datos BAO Reales (DESI)")
plt.plot(z_linea, DM_teorico_rd, 'r-', linewidth=2.5, label="Persistencia Hidrosocial (α=0.0096)")

plt.title('C. Oscilaciones de Bariones frente al Instrumento DESI')
plt.xlabel('Corrimiento al rojo (z)')
plt.ylabel('D_M(z) / r_d')
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend()
plt.tight_layout()
plt.show()
