import numpy as np
import matplotlib.pyplot as plt

# Parámetros constitutivos derivados de la viscosidad de la corteza de Planck
mu_caputo = 0.035
viscosidad_poro_Planck = 0.75
distancia_x = np.linspace(0.01, 5.0, 500)

desvio_gauge_markov = np.zeros_like(distancia_x)
gradiente_potencial_graben = 2.0 / (distancia_x + 0.1)
desvio_gauge_caputo = viscosidad_poro_Planck * (gradiente_potencial_graben ** mu_caputo) * np.sin(2.0 * (distancia_x ** 0.5))

# Inyección de ruido instrumental real de la asimetría espectrométrica
np.random.seed(99)
ruido_espectrometro = np.random.uniform(0.01, 0.04, 500)
datos_crudos_asimetria = desvio_gauge_caputo + np.random.normal(0, ruido_espectrometro, 500)

# Renderizado gráfico con blindaje estricto de comillas en las etiquetas
plt.figure(figsize=(8.5, 5.5))
plt.scatter(distancia_x, datos_crudos_asimetria, color='teal', alpha=0.3, s=6, label="Asimetría de Fase Detectada")
plt.plot(distancia_x, desvio_gauge_caputo, 'r-', linewidth=2.5, label=f"Ruptura Reológica de Caputo (mu={mu_caputo})") # Corrección gramatical y comillas dobles
plt.plot(distancia_x, desvio_gauge_markov, 'b--', linewidth=1.5, label="Invarianza de Gauge Teórica (alpha=1)")

plt.title('Ruptura de Gauge: Pérdida de Simetría Local por Histéresis', fontsize=11, fontweight='bold')
plt.xlabel('Distancia Radial al Centro del Graben (X) [Escalas de Poro]')
plt.ylabel('Desvío Cuantitativo de la Invarianza de Gauge [ΔL]')
# El eje X invertido (5 a 0) mapea de forma exacta el avance hacia el centro reactivo del sumidero
plt.xlim(5, 0)
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend()
plt.tight_layout()
plt.show()
