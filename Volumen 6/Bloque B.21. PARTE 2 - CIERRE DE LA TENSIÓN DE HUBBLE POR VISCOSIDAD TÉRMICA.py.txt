import numpy as np
import matplotlib.pyplot as plt

# Parámetros constitutivos derivados de la filtración de la cuenca abierta
mu_caputo = 0.035
T0_actual_k = 2.7255
z_redshift = np.linspace(0.0, 2.5, 500)

T_vacio_markov = T0_actual_k * (1.0 + z_redshift)
calor_ohmico_residual = 0.045 * (1.0 + z_redshift) ** (1.0 + mu_caputo)
T_vacio_persistencia = T_vacio_markov + calor_ohmico_residual

# La velocidad de expansión deducida directamente a partir de la fluidez del suelo cuántico
H0_acoplado_termico = 67.4 + (73.04 - 67.4) * (T0_actual_k / T_vacio_persistencia) ** (mu_caputo * 12.0)

# Renderizado gráfico con blindaje de comillas en las etiquetas
plt.figure(figsize=(8, 5.5))
plt.plot(z_redshift, H0_acoplado_termico, 'g-', linewidth=2.5, label="H0 deducido por Fluidez Térmica")
plt.axhline(y=73.04, color='crimson', linestyle=':', label="Meta Local SH0ES (73.04)")
plt.axhline(y=67.4, color='navy', linestyle=':', label="Meta Profunda Planck (67.4)")

plt.title('B. Cierre de la Tensión de Hubble por Viscosidad Térmica', fontsize=11, fontweight='bold')
plt.xlabel('Corrimiento al Rojo / Profundidad Temporal (z)')
plt.ylabel('Constante Extrapolada H_0(z) [km/s/Mpc]') # Corrección terminológica para blindaje científico
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend()
plt.tight_layout()
plt.show()
