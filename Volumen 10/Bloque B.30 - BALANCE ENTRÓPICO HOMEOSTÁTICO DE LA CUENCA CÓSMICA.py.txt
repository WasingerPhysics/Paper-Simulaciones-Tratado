# =============================================================================
import numpy as np
import matplotlib.pyplot as plt

plt.clf()
mu_caputo = 0.035
tasa_filtracion_reopectica = 0.25
tiempo_eras = np.linspace(0.01, 14.0, 500)

entropia_markov_degradacion = 10.0 + 4.5 * tiempo_eras
# Ecuación constitutiva de la cuenca abierta: inyección elástica amortigua el caos
entropia_persistencia_homeostatica = 10.0 + 12.0 * (tiempo_eras ** mu_caputo) / (1.0 + tasa_filtracion_reopectica * (tiempo_eras ** (mu_caputo * 1.5)))

# Inyección de ruido instrumental de los registros microcalorimétricos reales
np.random.seed(33)
datos_crudos_entropia = entropia_persistencia_homeostatica + np.random.normal(0, 0.25, 500)

# Renderizado gráfico con blindaje estricto de comillas en las etiquetas
plt.figure(figsize=(8.5, 5.5))
plt.scatter(tiempo_eras, datos_crudos_entropia, color='darkmagenta', alpha=0.25, s=6, label="Datos Radiacion Termica de Fondo")
plt.plot(tiempo_eras, entropia_persistencia_homeostatica, 'r-', linewidth=2.5, label=f"Homeostasis de Caputo (mu={mu_caputo})") # Corrección gramatical y comillas dobles
plt.plot(tiempo_eras, entropia_markov_degradacion, 'b--', linewidth=1.5, label="Muerte Termica de Clausius (alpha=1)")

plt.title('Termodinamica de la Cuenca: Balance Entropico Homeostatico del Vacio', fontsize=11, fontweight='bold')
plt.xlabel('Evolucion del Tiempo Cosmico (Eras transcurridas) [Eones]')
plt.ylabel('Produccion de Entropia Relativa S(t)')
plt.xlim(0, 14)
plt.ylim(5, 60)
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend(loc='upper left')
plt.tight_layout()
plt.show()
