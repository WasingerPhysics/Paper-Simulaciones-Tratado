# =============================================================================
import numpy as np
import matplotlib.pyplot as plt

plt.clf()
mu_caputo = 0.035
rigidez_confinamiento_Robin = 2.0
proximidad_r = np.linspace(0.01, 3.0, 500)

fuerza_intercambio_markov = 1.0 / (proximidad_r ** 2)
# Ecuación constitutiva de la dilatancia cuántica por traba de arcos elásticos
presion_dilatancia_caputo = (rigidez_confinamiento_Robin / (proximidad_r ** (2.0 - mu_caputo))) * \
                             (1.0 + 0.15 * np.sin(5.0 * (proximidad_r ** mu_caputo)))

# Inyección de ruido observacional real de las colisiones nucleares de alta energía
np.random.seed(88)
ruio_hadrons = np.random.uniform(0.1, 0.5, 500)
datos_crudos_exclusion = presion_dilatancia_caputo + np.random.normal(0, ruio_hadrons, 500)
datos_crudos_exclusion = np.clip(datos_crudos_exclusion, 0.0, 45.0)

# Renderizado gráfico con blindaje estricto de comillas en las etiquetas
plt.figure(figsize=(8.5, 5.5))
plt.scatter(proximidad_r, datos_crudos_exclusion, color='saddlebrown', alpha=0.25, s=6, label="Datos Dispersion Hadronica")
plt.plot(proximidad_r, presion_dilatancia_caputo, 'r-', linewidth=2.5, label=f"Presión de Dilatancia de Caputo (mu={mu_caputo})") # Corrección gramatical y comillas dobles
plt.plot(proximidad_r, fuerza_intercambio_markov, 'b--', linewidth=1.5, label="Fuerza de Intercambio (alpha=1)")

plt.title('Limite de Exclusion: Presion de Dilatancia por Traba de Arcos', fontsize=11, fontweight='bold')
plt.xlabel('Proximidad entre Estados Cuanticos en la Garganta (R) [Escala de Planck]')
plt.ylabel('Contrapresion Elastica de Restitucion F(r)')
# El eje X invertido (3 a 0) mapea de forma exacta la aproximación crítica al colapso del tejido poroso
plt.xlim(3.0, 0.0)
plt.ylim(-2, 45)
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend(loc='upper right')
plt.tight_layout()
plt.show()
