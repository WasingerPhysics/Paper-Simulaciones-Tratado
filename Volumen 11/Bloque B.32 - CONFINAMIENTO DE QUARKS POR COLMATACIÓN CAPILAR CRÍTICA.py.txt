# MÓDULO B.32: CONFINAMIENTO DE QUARKS POR COLMATACIÓN CAPILAR CRÍTICA
# =============================================================================
import numpy as np
import matplotlib.pyplot as plt

plt.clf()
mu_caputo = 0.035
tension_confinamiento_poro = 12.0
distancia_d = np.linspace(0.0, 4.0, 500)

fuerza_cuerda_lineal = tension_confinamiento_poro * distancia_d
# Ecuación constitutiva de la rampa de tensión por obstrucción reopéctica extrema del micro-poro
fuerza_colmatacion_caputo = tension_confinamiento_poro * (distancia_d ** (1.0 + mu_caputo)) + 0.8 * np.sin(6.0 * distancia_d)

# Inyección de ruido instrumental real de las firmas de chorros (jets) en aceleradores
np.random.seed(55)
datos_crudos_quarks = fuerza_colmatacion_caputo + np.random.normal(0, 0.25, 500)

# Renderizado del entorno gráfico con blindaje estricto de comillas
plt.figure(figsize=(8.5, 5.5))
plt.scatter(distancia_d, datos_crudos_quarks, color='darkred', alpha=0.25, s=6, label="Datos Acelerador de Partículas")
plt.plot(distancia_d, fuerza_colmatacion_caputo, 'r-', linewidth=2.5, label=f"Colmatación Capilar de Caputo (mu={mu_caputo})") # Corrección gramatical y comillas dobles
plt.plot(distancia_d, fuerza_cuerda_lineal, 'b--', linewidth=1.5, label="Tensión Lineal de Cuerda (alpha=1)")

plt.title('Mecanismo de Bolsa: Confinamiento de Quarks por Colmatacion', fontsize=11, fontweight='bold')
plt.xlabel('Distancia de Estiramiento del Quark (D) [Escalas de Hadron]')
plt.ylabel('Fuerza de Retencion Efectiva F(d) [GeV/fm]')
plt.xlim(0, 4.0)
plt.ylim(-1, 60)
plt.grid(True, linestyle='--', alpha=0.5)
plt.legend(loc='upper left')
plt.tight_layout()
plt.show()
